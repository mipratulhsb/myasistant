async submitToAI() { if (!this.userQuery.trim() || this.loadingAI) return; if (!cachedApiKey) { this.startTypingEffect("API Key belum ditemukan di database Supabase."); return; } const prompt = this.userQuery; this.userQuery = ''; this.loadingAI = true; this.typedText = ''; // Format payload yang bersih dan sesuai standar API Gemini const payload = { contents: [ { role: "user", parts: [ { text: `Profil Tuanmu: Nama ${this.profile.namaLengkap} (${this.profile.panggilan}). Wawasan/Instruksi: ${this.profile.customInstructions}\n\nPertanyaan: ${prompt}` } ] } ] }; try { const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${cachedApiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); const data = await response.json(); // Validasi jalur respons JSON yang mendalam if (data && data.candidates && data.candidates.length > 0 && data.candidates[0].content && data.candidates[0].content.parts && data.candidates[0].content.parts.length > 0) { const replyText = data.candidates[0].content.parts[0].text; this.startTypingEffect(replyText); } else { console.error("Struktur respons API tidak valid:", data); this.startTypingEffect("Maaf, format respons tidak dikenali server."); } } catch (error) { console.error("Error koneksi Gemini API:", error); this.startTypingEffect("Terjadi kesalahan koneksi ke server Gemini."); } finally { this.loadingAI = false; } }
